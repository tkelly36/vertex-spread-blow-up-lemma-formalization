\begin{definition}[Raw candidates]
For a blow-up instance $B$, a candidacy system $A$, a finite embedded
set $S$, and a total map $f:V(H)\to V(G)$, define
\[
 D_S^f(x)=\{v\in V(G):A(x,v),\quad
                  G(v,f(y))\text{ for every }y\in S\cap N_H(x)\}.
\]
This is a finite set, denoted $\operatorname{RawCandidates}(B,A,S,f,x)$.
In particular it does not remove the occupied images $f(S)$.
\end{definition}
