\begin{definition}
Let $W_1,W_2,W_3$ be pairwise disjoint finite vertex sets with $|W_1|=|W_2|=m$ and $|W_3|\in\{m-1,m,m+1\}$.  For $1\leq i<j\leq3$, let $(W_i,W_j)$ be $(\varepsilon,d_{ij})$-super-regular in $F_{ij}$ in the sense of \noderef{SuperRegularPair}.  The triple is a near-equal $(\varepsilon,d_{12},d_{13},d_{23})$-regular triplet if every edge $vu\in E(F_{12})$, with $v\in W_1$ and $u\in W_2$, satisfies
\[
 (d_{13}d_{23}-\varepsilon)|W_3|\leq
 |N_{F_{13}}(v)\cap N_{F_{23}}(u)\cap W_3|
 \leq(d_{13}d_{23}+\varepsilon)|W_3|.
\]
In particular, both the common-neighborhood window and all degree windows involving $W_3$ are normalized by the actual cardinality $|W_3|$, while the matching pair has the common size $m$.
\end{definition}
